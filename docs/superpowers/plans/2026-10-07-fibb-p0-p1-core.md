```ts
// gateway/src/ai/stream-readers.ts
/**
 * Pure body readers for streaming provider responses.
 * readNdjson: one JSON object per line (Ollama).
 * readSse: Server-Sent Events, `data:` lines hold JSON (Gemini alt=sse, OpenAI-compatible, Anthropic).
 * Both tolerate chunk boundaries anywhere and never throw on a bad line.
 */

export async function readNdjson(
  body: ReadableStream<Uint8Array>,
  onObject: (obj: any) => void,
): Promise<void> {
  const reader = body.getReader();
  const dec = new TextDecoder();
  let buf = '';
  for (;;) {
    const { value, done } = await reader.read();
    if (done) break;
    buf += dec.decode(value, { stream: true });
    let nl: number;
    while ((nl = buf.indexOf('\n')) >= 0) {
      const line = buf.slice(0, nl).trim();
      buf = buf.slice(nl + 1);
      if (!line) continue;
      try { onObject(JSON.parse(line)); } catch { /* skip malformed line */ }
    }
  }
  const tail = buf.trim();
  if (tail) { try { onObject(JSON.parse(tail)); } catch { /* ignore */ } }
}

export async function readSse(
  body: ReadableStream<Uint8Array>,
  onData: (obj: any) => void,
): Promise<void> {
  const reader = body.getReader();
  const dec = new TextDecoder();
  let buf = '';
  const handleEvent = (raw: string): boolean => {
    // Returns false when the stream should stop ([DONE]).
    const dataLines = raw.split('\n').filter(l => l.startsWith('data:')).map(l => l.slice(5).trim());
    if (dataLines.length === 0) return true;
    const data = dataLines.join('\n');
    if (data === '[DONE]') return false;
    try { onData(JSON.parse(data)); } catch { /* skip malformed event */ }
    return true;
  };
  for (;;) {
    const { value, done } = await reader.read();
    if (done) break;
    buf += dec.decode(value, { stream: true });
    let sep: number;
    while ((sep = buf.indexOf('\n\n')) >= 0) {
      const raw = buf.slice(0, sep);
      buf = buf.slice(sep + 2);
      if (!handleEvent(raw)) { try { await reader.cancel(); } catch { /* ignore */ } return; }
    }
  }
  if (buf.trim()) handleEvent(buf);
}
```

```ts
  /**
   * Streaming completion. Calls onChunk with each text delta as it arrives and
   * resolves with the same CompletionResponse shape as complete(). Providers
   * without a streaming path here (Claude) fall back to complete() and emit
   * the whole text as one chunk, so callers can always use this method.
   */
  async completeStream(
    request: CompletionRequest,
    onChunk: (text: string) => void,
  ): Promise<CompletionResponse> {
    const provider = this.providers.get(request.provider);
    if (!provider) throw new Error(`Provider ${request.provider} not found`);

    const emitOnce = async (): Promise<CompletionResponse> => {
      const res = await this.complete(request);
      onChunk(res.text);
      return res;
    };

    switch (provider.id) {
      case 'ollama':     return this.streamOllama(provider, request, onChunk);
      case 'gemini':     return this.streamGemini(provider, request, onChunk);
      case 'deepseek':   return this.streamOpenAICompatible(provider, request, 'deepseek_api_key', onChunk);
      case 'openai':     return this.streamOpenAICompatible(provider, request, 'openai_api_key', onChunk);
      case 'openrouter': return this.streamOpenAICompatible(provider, request, 'openrouter_api_key', onChunk);
      default:           return emitOnce();
    }
  }

  private async streamOllama(
    provider: AIProvider, request: CompletionRequest, onChunk: (t: string) => void,
  ): Promise<CompletionResponse> {
    let response: Response;
    try {
      response = await fetch(`${provider.endpoint}/api/chat`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          model: provider.model,
          messages: [{ role: 'system', content: request.system }, ...request.messages],
          stream: true,
          options: {
            temperature: request.temperature ?? 0.7,
            num_predict: request.maxTokens ?? provider.maxTokens,
          },
        }),
      });
    } catch (err: any) {
      throw new Error(`Ollama unreachable at ${provider.endpoint}: ${err?.message || err}. Is "ollama serve" running?`);
    }
    if (!response.ok || !response.body) {
      const body = await response.text().catch(() => '');
      throw new Error(`Ollama error ${response.status}: ${body.substring(0, 300) || response.statusText}`);
    }
    let text = '';
    let tokens = 0;
    await readNdjson(response.body, (obj) => {
      if (obj?.error) throw new Error(`Ollama error: ${obj.error}`);
      const delta: string = obj?.message?.content || '';
      if (delta) { text += delta; onChunk(delta); }
```

```ts
// gateway/src/services/chat-history.ts
/**
 * Per-channel chat history persisted as JSON under <memoryDir>/history/.
 * The MessagePipeline keeps its in-memory Map as the hot path; this store
 * makes it survive restarts (spec §3 "Persistent history").
 */
import { mkdir, readFile, writeFile } from 'fs/promises';
import { existsSync } from 'fs';
import { join } from 'path';

export interface HistoryMessage { role: string; content: string; timestamp: Date }

export class ChatHistoryStore {
  private dir: string;
  constructor(memoryDir: string, private maxMessages = 200) {
    this.dir = join(memoryDir, 'history');
  }

  private fileFor(channel: string): string {
    const safe = channel.replace(/[^a-zA-Z0-9_-]/g, '_').slice(0, 80);
    return join(this.dir, `${safe}.json`);
  }

  async load(channel: string): Promise<HistoryMessage[]> {
    const file = this.fileFor(channel);
    if (!existsSync(file)) return [];
    try {
      const raw = JSON.parse(await readFile(file, 'utf-8')) as Array<{ role: string; content: string; timestamp: string }>;
      return raw.map(m => ({ role: m.role, content: m.content, timestamp: new Date(m.timestamp) }));
    } catch {
      return [];
    }
  }

  async save(channel: string, messages: HistoryMessage[]): Promise<void> {
    await mkdir(this.dir, { recursive: true });
    const trimmed = messages.slice(-this.maxMessages).map(m => ({
      role: m.role, content: m.content, timestamp: m.timestamp.toISOString(),
    }));
    await writeFile(this.fileFor(channel), JSON.stringify(trimmed));
  }
}
```

// gateway/src/services/facts.ts
/**
 * Facts memory (spec §3). Durable, user-visible facts about the author and
 * their work, extracted after each chat turn by a cheap model call and
 * injected at the top of every system prompt. Distinct from user-model
 * (behavioral stats) and preferences (regex-detected style rules).
 */
import { mkdir, readFile, writeFile } from 'fs/promises';
import { existsSync } from 'fs';
import { join } from 'path';
import { logger } from './logger.js';

export interface FactBook { title: string; stage?: string; chapter?: number; goal?: string }
export interface FactPerson { name: string; kind: string; book?: string; notes?: string }
export interface Facts {
  name?: string;
  books: FactBook[];
  people: FactPerson[];
  preferences: string[];
  recent: string[];
  updatedAt?: string;
}
export type PartialFacts = Partial<Omit<Facts, 'updatedAt'>>;

export const EXTRACT_SYSTEM_PROMPT =
  'You extract durable facts about an author from one chat exchange. ' +
  'Return ONLY compact JSON with optional keys: name (string), books (array of {title, stage, chapter, goal}), ' +
  'people (array of {name, kind, book, notes}; kind is "character", "agent", "editor", "friend", etc.), ' +
  'preferences (array of short strings about how they like to work or write), ' +
  'recent (array of short strings about what they are doing or feeling right now). ' +
  'Omit keys you have no evidence for. Never invent. If nothing durable was said, return {}.';

const empty = (): Facts => ({ books: [], people: [], preferences: [], recent: [] });
const norm = (s: string) => s.trim().toLowerCase();

export class FactsStore {
  private file: string;
  private facts: Facts = empty();

  constructor(memoryDir: string) {
    this.file = join(memoryDir, 'facts.json');
  }

  async initialize(): Promise<void> {
    if (!existsSync(this.file)) return;
    try {
      const raw = JSON.parse(await readFile(this.file, 'utf-8'));
      this.facts = { ...empty(), ...raw };
    } catch (err) {
      logger.warn('facts.json unreadable, starting empty', err);
      this.facts = empty();
    }
  }

  get(): Facts {
    return JSON.parse(JSON.stringify(this.facts));
  }

  async merge(p: PartialFacts): Promise<void> {
    const f = this.facts;
    if (typeof p.name === 'string' && p.name.trim()) f.name = p.name.trim();

    for (const b of p.books || []) {
      if (!b?.title) continue;
      const hit = f.books.find(x => norm(x.title) === norm(b.title));
      if (hit) Object.assign(hit, stripUndefined({ stage: b.stage, chapter: b.chapter, goal: b.goal }));
      else f.books.push({ title: b.title.trim(), ...stripUndefined({ stage: b.stage, chapter: b.chapter, goal: b.goal }) });
    }

    for (const person of p.people || []) {
      if (!person?.name) continue;
      const hit = f.people.find(x => norm(x.name) === norm(person.name));
      if (hit) Object.assign(hit, stripUndefined({ kind: person.kind, book: person.book, notes: person.notes }));
      else f.people.push({ name: person.name.trim(), kind: person.kind || 'person', ...stripUndefined({ book: person.book, notes: person.notes }) });
    }

    for (const pref of p.preferences || []) {
      const s = String(pref).trim();
      if (s && !f.preferences.some(x => norm(x) === norm(s))) f.preferences.push(s);
    }
    f.preferences = f.preferences.slice(-20);

    const today = new Date().toISOString().slice(0, 10);
    for (const r of p.recent || []) {
      const s = String(r).trim();
      if (s) f.recent.push(`${today}: ${s}`);
    }
    f.recent = f.recent.slice(-10);

    f.updatedAt = new Date().toISOString();
    await this.persist();
  }

  /** Remove every fact whose text contains the phrase. Returns the count removed. */
  async forget(phrase: string): Promise<number> {
    const q = norm(phrase);
    if (!q) return 0;
    const f = this.facts;
    let n = 0;
    const keep = <T>(arr: T[], text: (t: T) => string): T[] =>
      arr.filter(t => { const hit = norm(text(t)).includes(q); if (hit) n++; return !hit; });
    f.books = keep(f.books, b => `${b.title} ${b.stage || ''} ${b.goal || ''}`);
    f.people = keep(f.people, p => `${p.name} ${p.kind} ${p.book || ''} ${p.notes || ''}`);
    f.preferences = keep(f.preferences, s => s);
    f.recent = keep(f.recent, s => s);
    if (f.name && norm(f.name).includes(q)) { delete f.name; n++; }
    if (n) await this.persist();
    return n;
  }

  /** Prompt block, capped at `budget` characters. */
  render(budget = 1500): string {
    const f = this.facts;
    const lines: string[] = ['# About the author'];
    if (!f.name) {
      lines.push("You don't know the author's name yet. In your next reply, warmly ask their name and what they're working on.");
    } else {
      lines.push(`Name: ${f.name}. Use it naturally.`);
    }
    if (f.books.length) {
      lines.push('Books: ' + f.books.map(b =>
        [b.title, b.stage, b.chapter ? `ch. ${b.chapter}` : '', b.goal ? `goal: ${b.goal}` : ''].filter(Boolean).join(' · ')).join('; '));
    }
    if (f.people.length) {
      lines.push('People: ' + f.people.map(p =>
        [p.name, p.kind, p.book, p.notes].filter(Boolean).join(', ')).join('; '));
    }
    if (f.preferences.length) lines.push('Preferences: ' + f.preferences.join('; '));
    if (f.recent.length) lines.push('Recently: ' + f.recent.slice(-5).join('; '));
    let out = lines.join('\n');
    if (out.length > budget) out = out.slice(0, budget - 1) + '…';
    return out;
  }

  /**
   * Run the extractor over one exchange. `complete` is any function that takes
   * (system, user) and returns the model's text; the pipeline passes a thin
   * wrapper around aiRouter.complete on the cheapest provider.
   */
  async extractFromTurn(
    userMessage: string,
    assistantReply: string,
    complete: (system: string, user: string) => Promise<string>,
  ): Promise<void> {
    const user = `USER SAID:\n${userMessage.slice(0, 3000)}\n\nASSISTANT REPLIED:\n${assistantReply.slice(0, 1500)}`;
    let text: string;
    try { text = await complete(EXTRACT_SYSTEM_PROMPT, user); } catch (err) { logger.debug('facts extract failed', err); return; }
    const parsed = parseJsonLoose(text);
    if (!parsed || typeof parsed !== 'object') return;
    await this.merge(parsed as PartialFacts);
  }

  private async persist(): Promise<void> {
    await mkdir(join(this.file, '..'), { recursive: true });
    await writeFile(this.file, JSON.stringify(this.facts, null, 2));
  }
}

function stripUndefined<T extends Record<string, any>>(o: T): Partial<T> {
  const out: any = {};
  for (const [k, v] of Object.entries(o)) if (v !== undefined && v !== null && v !== '') out[k] = v;
  return out;
}

/** Accepts raw JSON, a ```json fence, or JSON embedded in prose. Returns null on failure. */
export function parseJsonLoose(text: string): any | null {
  if (!text) return null;
  const fence = text.match(/```(?:json)?\s*([\s\S]*?)```/i);
  const candidates = [fence?.[1], text, text.slice(text.indexOf('{'), text.lastIndexOf('}') + 1)];
  for (const c of candidates) {
    if (!c) continue;
    try { return JSON.parse(c); } catch { /* try next */ }
  }
  return null;
}

```ts
// gateway/src/api/routes/chat-stream.ts
/**
 * POST /api/chat/stream — Server-Sent Events chat. Events:
 *   data: {"type":"chunk","content":"..."}   repeated
 *   data: {"type":"done","content":"<full text>"}
 *   data: {"type":"error","message":"..."}
 * GET /api/facts, DELETE /api/facts?phrase=... — the facts memory.
 */
import type { Request, Response } from 'express';
import type { ApiContext } from '../context.js';

export function registerChatStreamRoutes(ctx: ApiContext): void {
  const { app, gateway, services } = ctx;

  app.post('/api/chat/stream', async (req: Request, res: Response) => {
    const { message, channel } = req.body || {};
    if (!message || typeof message !== 'string') return res.status(400).json({ error: 'Message required' });
    if (message.length > 10000) return res.status(400).json({ error: 'Message too long (max 10,000 chars)' });
    const ch = typeof channel === 'string' && /^[a-zA-Z0-9:_-]{1,64}$/.test(channel) ? channel : 'webchat';

    res.setHeader('Content-Type', 'text/event-stream');
    res.setHeader('Cache-Control', 'no-cache');
    res.setHeader('Connection', 'keep-alive');
    res.flushHeaders();
    const send = (obj: unknown) => { res.write(`data: ${JSON.stringify(obj)}\n\n`); };

    try {
      await gateway.handleMessage(
        message, ch,
        (full: string) => send({ type: 'done', content: full }),
        undefined, undefined, undefined,
        (chunk: string) => send({ type: 'chunk', content: chunk }),
      );
    } catch (err: any) {
      const msg = String(err?.message || err);
      send({ type: 'error', message: msg.includes('No AI providers')
        ? 'No AI provider is reachable. Paste a free Gemini key or install Ollama.'
        : 'AI error: ' + msg });
    }
    res.end();
  });

  app.get('/api/facts', (_req: Request, res: Response) => {
    res.json(services.facts.get());
  });

  app.delete('/api/facts', async (req: Request, res: Response) => {
    const phrase = String(req.query.phrase || '');
    if (!phrase) return res.status(400).json({ error: 'phrase required' });
    const removed = await services.facts.forget(phrase);
    res.json({ removed });
  });
}
```

```ts
export type StreamEvent =
  | { type: 'chunk'; content: string }
  | { type: 'done'; content: string }
  | { type: 'error'; message: string };

export async function* streamChat(message: string, channel = 'webchat'): AsyncGenerator<StreamEvent> {
  const res = await fetch('/api/chat/stream', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ message, channel }),
  });
  if (!res.ok || !res.body) {
    let msg = `HTTP ${res.status}`;
    try { msg = (await res.json()).error || msg; } catch { /* ignore */ }
    yield { type: 'error', message: msg };
    return;
  }
  const reader = res.body.getReader();
  const dec = new TextDecoder();
  let buf = '';
  for (;;) {
    const { value, done } = await reader.read();
    if (done) break;
    buf += dec.decode(value, { stream: true });
    let sep: number;
    while ((sep = buf.indexOf('\n\n')) >= 0) {
      const raw = buf.slice(0, sep); buf = buf.slice(sep + 2);
      const data = raw.split('\n').filter(l => l.startsWith('data:')).map(l => l.slice(5).trim()).join('\n');
      if (!data) continue;
      try { yield JSON.parse(data) as StreamEvent; } catch { /* skip */ }
    }
  }
}
```

```tsx
    for await (const ev of streamChat(text)) {
      if (ev.type === 'chunk') { acc += ev.content; setMsgs(m => m.map(x => x.id === aid ? { ...x, content: acc } : x)); }
      else if (ev.type === 'done') { acc = ev.content || acc; setMsgs(m => m.map(x => x.id === aid ? { ...x, content: acc, streaming: false } : x)); }
      else if (ev.type === 'error') {
        if (/no ai provider/i.test(ev.message)) setNeedsProvider(true);
        setMsgs(m => m.map(x => x.id === aid ? { ...x, content: ev.message, streaming: false } : x));
      }
    }
    setBusy(false);
    onTurn();
  };

  const chips = msgs.length <= 1
    ? ["I'm working on a novel", 'Help me find a premise', "What can you do?"]
    : ['Check the last chapter', "What's on tonight?", 'Brainstorm a scene'];

  return (
    <>
      <div className="thread">
        <div className="thread-inner">
          <div className="day">Today</div>
          {msgs.map(m => <Message key={m.id} m={m} />)}
          {needsProvider && <FirstRunCard onDone={() => { setNeedsProvider(false); send('Thanks, you’re connected now. Say hi?'); }} />}
          <div ref={endRef} />
        </div>
      </div>
      <div className="chips">
        {chips.map(c => <button key={c} className="chip" onClick={() => send(c)} disabled={busy}>{c}</button>)}
      </div>
      <Composer onSend={send} disabled={busy} />
    </>
  );
}
```
